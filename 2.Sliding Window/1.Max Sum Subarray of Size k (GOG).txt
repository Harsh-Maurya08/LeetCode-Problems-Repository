class Solution {
    public int maxSubarraySum(int[] arr, int k) {
        int low=0;
        int high=k-1;
        int sum=0;
        for(int i=0;i<=high;i++){
            sum+=arr[i];
        }
        int res=sum;
        high++;
        low++;
        while(high<arr.length){
            sum = sum-arr[low-1]+arr[high];
            if(sum>res){
                res=sum;
            }
            high++;
            low++;
        }
        return res;
    }
}
