class Solution {
    int countTriplets(int sum, int arr[]) {
        Arrays.sort(arr);
        int count=0;
        for(int i=0;i<arr.length-2;i++){
         int j=i+1;
         int k= arr.length-1;
         while(j<k){
             int tri = arr[i]+arr[j]+arr[k];
             if(tri<sum){
                 count += (k-j);
                 j++;
             }
             else{
                 k--;
             }
         }
        }
        return count;
    }
}
