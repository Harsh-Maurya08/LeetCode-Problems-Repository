class Solution {
    public int removeDuplicates(int[] nums) {
        int res=1;
        if(nums.length==0)[
            res=0;
            return res;
        ]
        int i = 0;
        int j = 1;
        while(j<nums.length){
            if(nums[i]!=nums[j]){
                res++;
                nums[i+1]=nums[j];
                i++;
            }
            j++;
        }
        return res;
    }
}
