class Solution {
    void segregate0and1(int[] arr) {
        int i=0;
        int j=1;
        while(j<=arr.length-1){
            if(arr[i]!=0){
                if(arr[i]!=arr[j]){
                    int temp = arr[i];
                    arr[i] = arr[j];
                    arr[j]=temp;
                    i++;
                    j++;
                }
                
                else{
                  j++;
                }
            }
            else{
              i++;
              j++;
            }
        }
    }
}
